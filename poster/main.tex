\begin{block}{Problem \& Motivation}
\begin{itemize}
  \item NASA Artemis program: return humans to Moon by 2026
  \item Current rovers rely on Earth-based teleoperation
  \item Communication delay ($\sim$1.3\,s Earth--Moon) limits real-time control
  \item Need: fully autonomous navigation in unstructured terrain
  \item Our approach: \textbf{train in simulation, deploy on physical hardware}
\end{itemize}

\begin{alertblock}{Key Contributions}
\begin{itemize}
  \item \textbf{Sim-to-real transfer} of RL navigation onto small-scale rover
  \item \textbf{Real-time terrain segmentation} on Jetson edge hardware
  \item \textbf{Three-tier safety system} with obstacle memory
  \item \textbf{Web-based mission control} for remote monitoring
  \item \textbf{3D coordinate tracking} with active gimbal compensation
\end{itemize}

\begin{block}{Challenges \& Lessons}
\begin{itemize}
  \item \textbf{Stereo dead zone:} OAK-D Lite returns 0 depth $<$20\,cm\\
        \textit{Solved:} 85\% zero-pixel threshold triggers emergency protocol
  \item \textbf{Sim-to-real gap:} Simulation sensors $\ne$ physical camera\\
        \textit{Solved:} Depth-only observations, visual task variant
  \item \textbf{Safety vs.\ autonomy:} Aggressive thresholds $\to$ excessive caution\\
        \textit{Solved:} Iterative real-world tuning across 4+ rounds
  \item \textbf{API breaks:} DepthAI v2$\to$v3 broke camera initialization\\
        \textit{Solved:} Complete pipeline rewrite for v3 API
\end{itemize}
\end{block}

\begin{block}{Related Work}
\begin{itemize}
  \item \textbf{NASA VIPER:} Teleoperated, no onboard autonomy
  \item \textbf{Perseverance AutoNav:} Classical path planning, no RL
  \item \textbf{Stanford/CMU sim-to-real:} Large platforms, server-grade compute
  \item \textbf{Our novelty:} Full RL+CV pipeline on \textit{small-scale edge hardware}
\end{itemize}
\end{block}

\begin{block}{Future Work}
\begin{itemize}

  \item Unity simulation with ML object detection as RL input
